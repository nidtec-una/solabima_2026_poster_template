% -------------------------------------------------------------
% Estilo de bloque: secciones "en caja" (barra de titulo +
% cuerpo con borde), fondo de pagina blanco, acentos con la
% paleta SOLABIMA. Solo dos esquinas redondeadas -- superior
% izquierda e inferior derecha -- el resto quedan en angulo
% recto (ver \SolabimaR para el radio).
% -------------------------------------------------------------
\def\SolabimaR{20pt}
\defineblockstyle{SolabimaBox}{
    titlewidthscale=1, bodywidthscale=1, titleleft,
    titleoffsetx=0pt, titleoffsety=0pt, bodyoffsetx=0pt, bodyoffsety=0pt,
    bodyverticalshift=0pt, roundedcorners=0, linewidth=0.12cm,
    titleinnersep=0.9cm, bodyinnersep=1.15cm
}{
    \begin{scope}[line width=\blocklinewidth]
        \ifBlockHasTitle
           % Contorno completo (titulo+cuerpo): redondea la
           % esquina superior izquierda (del titulo) y la
           % inferior derecha (del cuerpo), igual que el relleno
           % de mas abajo, para que ambos trazos coincidan
           % exactamente y no asome una esquina recta detras del
           % redondeo.
           \draw[color=blocktitlebgcolor, fill=blocktitlebgcolor]
             ($(blocktitle.north west)+(\SolabimaR,0)$)
             -- (blocktitle.north east)
             -- ($(blockbody.south east)+(0,\SolabimaR)$)
             arc[start angle=0, end angle=-90, radius=\SolabimaR]
             -- (blockbody.south west)
             -- ($(blocktitle.north west)+(0,-\SolabimaR)$)
             arc[start angle=180, end angle=90, radius=\SolabimaR]
             -- cycle;
           % Relleno del cuerpo: redondea solo la esquina
           % inferior derecha.
           \draw[color=blocktitlebgcolor, fill=blockbodybgcolor]
             (blockbody.south west)
             -- ($(blockbody.south east)+(-\SolabimaR,0)$)
             arc[start angle=-90, end angle=0, radius=\SolabimaR]
             -- (blockbody.north east)
             -- (blockbody.north west)
             -- cycle;
        \else
           % Sin titulo (p.ej. franja de logos): un solo
           % rectangulo, redondea superior izquierda e inferior
           % derecha.
           \draw[color=blocktitlebgcolor, fill=blockbodybgcolor]
             ($(blockbody.north west)+(\SolabimaR,0)$)
             -- (blockbody.north east)
             -- ($(blockbody.south east)+(0,\SolabimaR)$)
             arc[start angle=0, end angle=-90, radius=\SolabimaR]
             -- (blockbody.south west)
             -- ($(blockbody.north west)+(0,-\SolabimaR)$)
             arc[start angle=180, end angle=90, radius=\SolabimaR]
             -- cycle;
        \fi
    \end{scope}
}
\useblockstyle{SolabimaBox}
